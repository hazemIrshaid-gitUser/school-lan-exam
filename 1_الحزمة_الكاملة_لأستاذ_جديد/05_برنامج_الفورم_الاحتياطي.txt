/*
 ==========================================================================================
  برنامج الفورم الاحتياطي: بيعمل Google Form (اختبار بيصحّح لحاله) من ملف الأسئلة نفسه.
  الخطوات بالتفصيل والصور بملف 05_الفورم_الاحتياطي.pdf. باختصار:
   1) ارفع questions.xlsx على Google Drive (ومجلد images جنبه إذا في صور)، وافتحه،
      ومن «ملف» اختار «حفظ كـ Google Sheets».
   2) بالملف الجديد: «الإضافات» ← «Apps Script». امسح كل اللي بالصفحة، والصق هذا الملف كله، واضغط حفظ.
   3) فوق اختار makeForm واضغط «تشغيل»، ووافق على الصلاحيات (أول مرة بس).
   4) ارجع لملف الأسئلة: الروابط بورقة جديدة اسمها «روابط الفورم».
  الفورم بيطلع مسكّر (ما بيستقبل إجابات) لحد ما تفتحه وقت الطوارئ.
 ==========================================================================================
*/

// ============ الإعدادات: غيّرها إذا بدك (الفاضي "" = الافتراضي) ============
var SETTINGS = {
  title: "",             // عنوان الامتحان. فاضي = اسم ملف الأسئلة
  minutes: "",           // مدة الامتحان بالدقائق: بتنكتب للطالب بأول الفورم بس (الفورم ما فيه مؤقت: سكّره بإيدك)
  hard: "",              // كم سؤال صعب. فاضي = كل الأسئلة الصعبة بالملف
  medium: "",            // كم سؤال وسط. فاضي = كلها
  easy: "",              // كم سؤال سهل. فاضي = كلها
  total: "",             // إذا عمود المستوى فاضي بالملف كله: كم سؤال. فاضي = كل الأسئلة
  questionMark: "",      // علامة واحدة لكل سؤال (بتغلب عمود العلامة). فاضي = من الملف (والفاضي هناك = 1)
  versions: 1,           // كم نموذج (1 إلى 10): 1 = فورم واحد. 2 = نموذجين بأسئلة مختلفة (لما الملف فيه أسئلة أكثر من المطلوب)
  shuffleOptions: true   // خلط ترتيب الخيارات (true = نعم، false = لا)
};

// من هون وتحت لا تغيّر إشي

var LINKS_SHEET = "روابط الفورم";
var LET_IN = {"أ": 0, "ا": 0, "إ": 0, "a": 0, "1": 0, "ب": 1, "b": 1, "2": 1, "ج": 2, "c": 2, "3": 2, "د": 3, "d": 3, "4": 3};
var LVL_IN = {"صعب": "hard", "صعبة": "hard", "hard": "hard", "وسط": "medium", "متوسط": "medium", "متوسطة": "medium", "medium": "medium",
              "سهل": "easy", "سهلة": "easy", "easy": "easy", "": ""};
var LVL_AR = {hard: "صعب", medium: "وسط", easy: "سهل"};
var YES = ["نعم", "yes", "true", "1", "y", "نعم.", "ايوا", "أجل"], NO = ["لا", "no", "false", "0", "n", "لا."];
var IMG_RE = /\[\s*(?:img|IMG|Img|صورة)\s*[:：]\s*([^\]\n]+?)\s*\]/g;
var LRI = "⁦", PDI = "⁩";
var MATH_RUN = /[A-Za-z0-9Ͱ-Ͽ←-⋿°π√∞∑∫±×÷−=<>≤≥≠≈^_\/|()\[\]{}⁰-₟²³¹.,:;%+\-'’ ]+/g;

function makeForm() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (!ss) throw new Error("شغّل البرنامج من داخل ملف الأسئلة: افتح الملف بـ Google Sheets ثم «الإضافات» ← «Apps Script».");
  var sheet = findQuestionSheet(ss);
  var p = parseQuestions(sheet.getDataRange().getDisplayValues());
  if (p.errs.length) return fail(ss, p.errs);
  if (!p.qs.length) return fail(ss, ["ملف الأسئلة فاضي" + (p.warns.length ? " (الأسئلة اللي فيه كلها فيها صور بالخيارات)" : "")]);
  if (given(SETTINGS.questionMark)) {
    var qm = toNum(SETTINGS.questionMark);
    if (!(qm > 0 && qm <= 100)) return fail(ss, ["الإعداد questionMark لازم يكون رقم أكبر من 0، لقيت «" + SETTINGS.questionMark + "»"]);
    p.qs.forEach(function (q) { q.marks = qm; });
  }
  var pk = computePick(p.qs, SETTINGS);
  if (pk.errs.length) return fail(ss, pk.errs);
  var nv = given(SETTINGS.versions) ? toNum(SETTINGS.versions) : 1;
  if (!(nv >= 1 && nv <= 10 && nv % 1 === 0)) return fail(ss, ["الإعداد versions لازم يكون من 1 إلى 10، لقيت «" + SETTINGS.versions + "»"]);
  var papers = [];
  for (var v = 0; v < nv; v++) papers.push(makePaper(p.qs, pk.pick, Math.random));
  var names = [];
  papers.forEach(function (pp) { pp.forEach(function (q) { q.imgs.forEach(function (n) { if (names.indexOf(n) < 0) names.push(n); }); }); });
  var im = loadImages(ss, names);
  if (im.errs.length) return fail(ss, im.errs);
  var all = [];
  papers.forEach(function (pp) { pp.forEach(function (q) { all.push(q.marks); }); });
  var scale = pointsScale(all);
  var title = String(SETTINGS.title || "").trim() || ss.getName().replace(/\.xlsx?$/i, "");
  var made = papers.map(function (pp, i) { return buildForm(ss, title, pp, i, nv, scale, im.blobs, Math.random); });
  var notes = p.warns.concat(pk.warns);
  if (scale !== 1) notes.push("الفورم ما بيقبل علامات بكسور، فكل العلامات انضربت بـ " + scale + " (العلامة الحقيقية = علامة الفورم ÷ " + scale + ").");
  writeLinks(ss, made, notes);
  var msg = "تم ✓ عملت " + (nv === 1 ? "الفورم" : nv === 2 ? "نموذجين" : nv + " نماذج") + "، والروابط بورقة «" + LINKS_SHEET + "» بملف الأسئلة.\n\n" + MANUAL_STEPS +
            (notes.length ? "\n\nملاحظات:\n• " + notes.join("\n• ") : "");
  say(msg);
}

function onOpen() {
  SpreadsheetApp.getUi().createMenu("📝 الفورم الاحتياطي").addItem("إنشاء الفورم من الأسئلة", "makeForm").addToUi();
}

var MANUAL_STEPS =
  "قبل الامتحان، لكل فورم:\n" +
  "1) افتح «رابط التعديل» ← «الإعدادات» ← قسم الاختبار ← «يمكن للمجيبين الاطلاع على»: تأكد إنه «الأسئلة الفائتة» و«الإجابات الصحيحة»" +
  " (Missed questions و Correct answers) مش معلّمين ✓. Google بتطلعهم مطفيين، وإذا لقيتهم معلّمين شيلهم: بدونها اللي بيسلّم أول بيشوف الإجابات وبيبعتها لزملائه.\n" +
  "2) جرّب الفورم من زر المعاينة 👁 وتأكد من الأسئلة والإجابات.\n" +
  "الفورم مسكّر هلأ. وقت الطوارئ: «الردود» ← شغّل «قبول الردود».";

// ---------- قراءة الأسئلة: نفس قواعد server.py ----------
function own(o, k) { return Object.prototype.hasOwnProperty.call(o, k); }
function given(v) { return String(v == null ? "" : v).trim() !== ""; }
function num(x) { return String(Math.round(x * 100) / 100); }
function toNum(s) {
  var t = String(s).trim().replace(/[٠-٩]/g, function (d) { return "٠١٢٣٤٥٦٧٨٩".indexOf(d); }).replace(/[٫،,]/g, ".");
  return /^\d+(\.\d+)?$/.test(t) ? parseFloat(t) : NaN;
}
function lostText(t) {
  var s = String(t).replace(/\s/g, ""), n = s.split("?").length - 1;
  return s.length >= 3 && n >= 3 && n / s.length >= 0.6;
}
function autoLtr(t) {
  // المقاطع الرياضية/اللاتينية جوّا جملة عربية بتنعزل من اليسار لليمين عشان ما تنعكس (زي السيرفر)
  if (t.indexOf(LRI) >= 0 || !/[؀-ۿ]/.test(t)) return t;
  return t.replace(MATH_RUN, function (s) {
    var lead = s.match(/^[ .,:;]*/)[0], core = s.slice(lead.length).replace(/[ .,:;]+$/, ""), tail = s.slice(lead.length + core.length);
    return /[A-Za-z0-9Ͱ-Ͽ°√∞]/.test(core) ? lead + LRI + core + PDI + tail : s;
  });
}
function findQuestionSheet(ss) {
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sh = sheets[i];
    if (sh.getName() === LINKS_SHEET || sh.getLastColumn() < 1) continue;
    var head = sh.getRange(1, 1, 1, sh.getLastColumn()).getDisplayValues()[0].join(" ").toLowerCase();
    if (head.indexOf("السؤال") >= 0 || head.indexOf("question") >= 0) return sh;
  }
  return sheets[0];
}
function parseQuestions(rows) {
  var qs = [], errs = [], warns = [];
  for (var i = 1; i < rows.length; i++) {
    var line = i + 1, r = rows[i].map(function (c) { return String(c == null ? "" : c).trim(); });
    while (r.length < 11) r.push("");
    if (!r.some(function (c) { return c; })) continue;
    var q = r[0], opts = r.slice(1, 5), ans = r[5], lvl = r[6].toLowerCase(), mk = r[9], fx = r[10].toLowerCase(), marks = 1;
    if (mk) {
      marks = toNum(mk);
      if (isNaN(marks)) { errs.push("سطر " + line + ": عمود «العلامة» لازم يكون رقم (مثلاً 1 أو 2 أو 0.5)، لقيت «" + mk + "»"); continue; }
      if (!(marks > 0 && marks <= 100)) { errs.push("سطر " + line + ": العلامة لازم تكون أكبر من 0 وأقل من أو تساوي 100، لقيت «" + mk + "»"); continue; }
    }
    if (!q) { errs.push("سطر " + line + ": نص السؤال فاضي"); continue; }
    if (lostText(q) || opts.some(lostText)) { errs.push("سطر " + line + ": النص علامات استفهام (؟؟؟): الملف انحفظ بترميز ما بيدعم العربي"); continue; }
    var k = opts.length;
    while (k && !opts[k - 1]) k--;
    opts = opts.slice(0, k);
    if (k < 2 || opts.some(function (o) { return !o; })) { errs.push("سطر " + line + ": لازم خيارين على الأقل بدون فراغ بينهم"); continue; }
    var key = ans.toLowerCase().replace(/\./g, ""), a = own(LET_IN, key) ? LET_IN[key] : -1;
    if (a < 0 || a >= k) { errs.push("سطر " + line + ": عمود «الجواب» لازم يكون أ أو ب أو ج أو د (وضمن عدد الخيارات)، لقيت «" + ans + "»"); continue; }
    if (fx && YES.indexOf(fx) < 0 && NO.indexOf(fx) < 0) { errs.push("سطر " + line + ": عمود «ثابت» لازم يكون نعم أو فاضي، لقيت «" + r[10] + "»"); continue; }
    if (!own(LVL_IN, lvl)) { errs.push("سطر " + line + ": المستوى لازم يكون صعب أو وسط أو سهل (أو فاضي)، لقيت «" + r[6] + "»"); continue; }
    if (opts.some(function (o) { IMG_RE.lastIndex = 0; return IMG_RE.test(o); })) {
      warns.push("سطر " + line + ": فيه صورة داخل الخيارات، والفورم ما بيدعم هيك، فالسؤال مش موجود بالفورم");
      continue;
    }
    var seen = {}, dup = opts.some(function (o) { if (own(seen, o)) return true; seen[o] = 1; return false; });
    if (dup) { errs.push("سطر " + line + ": في خيارين متطابقين، والفورم ما بيقبل خيارات مكررة"); continue; }
    var imgs = [];
    var text = q.replace(IMG_RE, function (m, name) { imgs.push(name.trim()); return " "; }).replace(/ {2,}/g, " ").trim();
    qs.push({row: line, q: autoLtr(text), o: opts.map(autoLtr), a: a, level: LVL_IN[lvl], marks: marks, fixed: YES.indexOf(fx) >= 0, imgs: imgs});
  }
  return {qs: qs, errs: errs, warns: warns};
}
function computePick(qs, s) {
  var errs = [], warns = [], pick, cnt = {hard: 0, medium: 0, easy: 0, "": 0};
  qs.forEach(function (q) { cnt[q.level]++; });
  if (qs.every(function (q) { return q.level === ""; })) {              // ما في مستويات: العدد الكلي من الكل
    qs.forEach(function (q) { q.level = "medium"; });
    var n = given(s.total) ? toNum(s.total) : qs.length;
    if (!(n >= 1 && n % 1 === 0)) return {errs: ["الإعداد total لازم يكون رقم صحيح، لقيت «" + s.total + "»"], warns: warns};
    if (n > qs.length) warns.push("عدد الأسئلة المطلوب (" + n + ") أكبر من الملف (" + qs.length + ")، فانحطّوا كلهم");
    pick = {hard: 0, medium: Math.min(n, qs.length), easy: 0};
  } else {
    var empty = qs.filter(function (q) { return q.level === ""; });
    if (empty.length) errs.push("سطر " + empty[0].row + ": المستوى فاضي، وباقي الأسئلة إلها مستوى. عبّيه أو امسح عمود المستوى كله");
    pick = {};
    ["hard", "medium", "easy"].forEach(function (k) {
      var v = given(s[k]) ? toNum(s[k]) : cnt[k];
      if (!(v >= 0 && v % 1 === 0)) { errs.push("الإعداد " + k + " لازم يكون رقم صحيح، لقيت «" + s[k] + "»"); v = 0; }
      if (v > cnt[k]) errs.push("أسئلة الملف أقل من المطلوب: " + LVL_AR[k] + " مطلوب " + v + " والموجود " + cnt[k]);
      pick[k] = v;
    });
  }
  ["hard", "medium", "easy"].forEach(function (k) {
    var f = qs.filter(function (q) { return q.level === k && q.fixed; }).length;
    if (f > pick[k]) errs.push("الأسئلة الثابتة (عمود «ثابت») أكثر من المطلوب: " + (LVL_AR[k]) + " ثوابت " + f + " والمطلوب " + pick[k]);
  });
  if (!errs.length && pick.hard + pick.medium + pick.easy === 0) errs.push("عدد الأسئلة بالفورم صفر: راجع الإعدادات فوق");
  return {pick: pick, errs: errs, warns: warns};
}
function shuffled(arr, rnd) {
  var a = arr.slice();
  for (var i = a.length - 1; i > 0; i--) { var j = Math.floor(rnd() * (i + 1)), t = a[i]; a[i] = a[j]; a[j] = t; }
  return a;
}
function makePaper(qs, pick, rnd) {
  // زي السيرفر: الثابتة بكل نموذج، والباقي سحب عشوائي من كل مستوى، وبعدين ترتيب عشوائي
  var ids = [];
  ["hard", "medium", "easy"].forEach(function (lv) {
    var fixed = qs.filter(function (q) { return q.level === lv && q.fixed; }), rest = qs.filter(function (q) { return q.level === lv && !q.fixed; });
    ids = ids.concat(fixed, shuffled(rest, rnd).slice(0, pick[lv] - fixed.length));
  });
  return shuffled(ids, rnd);
}
function pointsScale(marks) {
  // علامات الفورم أرقام صحيحة بس: 0.5 بتصير 1 لما نضرب الكل بـ 2
  var fs = [1, 2, 4, 5, 10, 20, 100];
  for (var i = 0; i < fs.length; i++) {
    var f = fs[i];
    if (marks.every(function (m) { return Math.abs(m * f - Math.round(m * f)) < 1e-9; })) return f;
  }
  return 100;
}

// ---------- Google: الصور والفورم وورقة الروابط ----------
function loadImages(ss, names) {
  var blobs = {}, errs = [];
  if (!names.length) return {blobs: blobs, errs: errs};
  var dirs = [], parents = DriveApp.getFileById(ss.getId()).getParents();
  while (parents.hasNext()) {
    var it = parents.next().getFoldersByName("images");
    while (it.hasNext()) dirs.push(it.next());
  }
  if (!dirs.length) return {blobs: blobs, errs: ["الأسئلة فيها صور، بس ما في مجلد اسمه images جنب ملف الأسئلة على Google Drive. ارفع مجلد images لنفس المجلد وشغّل البرنامج مرة ثانية."]};
  names.forEach(function (n) {
    for (var i = 0; i < dirs.length; i++) {
      var f = dirs[i].getFilesByName(n);
      if (f.hasNext()) { blobs[n] = f.next().getBlob(); return; }
    }
    errs.push("الصورة «" + n + "» مش موجودة بمجلد images على Google Drive");
  });
  return {blobs: blobs, errs: errs};
}
function buildForm(ss, title, paper, vi, nv, scale, blobs, rnd) {
  var name = nv > 1 ? title + " (نموذج " + "أبجدهوزحطي".charAt(vi) + ")" : title;
  var total = 0;
  paper.forEach(function (q) { total += Math.round(q.marks * scale); });
  var form = FormApp.create(name);
  form.setIsQuiz(true);
  form.setDescription("امتحان احتياطي" + (given(SETTINGS.minutes) ? "، مدته " + SETTINGS.minutes + " دقيقة" : "") + ".\n" +
                      "• اكتب اسمك الكامل وصفّك وشعبتك، واضغط «التالي».\n" +
                      "• اختر إجابة وحدة لكل سؤال. بعد ما تخلص اضغط «إرسال» مرة وحدة.\n" +
                      "• عدد الأسئلة: " + paper.length + "، والعلامة من " + num(total / scale) +
                      (scale !== 1 ? " (الفورم بيعرضها من " + total + "، يعني مضروبة بـ " + scale + ")" : "") + ".");
  form.setProgressBar(true);
  form.setShowLinkToRespondAgain(false);
  form.setConfirmationMessage("تم التسليم ✓");
  form.addTextItem().setTitle("الاسم الكامل").setRequired(true);
  form.addTextItem().setTitle("الصف والشعبة").setHelpText("مثلاً: العاشر أ").setRequired(true);
  form.addPageBreakItem().setTitle("الأسئلة");
  var hasImg = false;
  paper.forEach(function (q) {
    q.imgs.forEach(function (n) { hasImg = true; form.addImageItem().setImage(blobs[n]); });
    var item = form.addMultipleChoiceItem(), order = q.o.map(function (o, j) { return j; });
    if (SETTINGS.shuffleOptions !== false) order = shuffled(order, rnd);
    item.setTitle(q.q || "اختر الإجابة الصحيحة حسب الصورة")
        .setChoices(order.map(function (j) { return item.createChoice(q.o[j], j === q.a); }))
        .setPoints(Math.round(q.marks * scale))
        .setRequired(false);                 // مش إجباري: لما يخلص الوقت الطالب بيسلّم حتى لو ما حلّ الكل
  });
  form.setShuffleQuestions(!hasImg);          // الخلط بيفصل الصورة عن سؤالها، فمع الصور الترتيب عشوائي بس ثابت للنموذج
  form.setAcceptingResponses(false);
  try { form.setCustomClosedFormMessage("الامتحان الاحتياطي مش مفتوح هلأ. استنّى تعليمات الأستاذ."); }
  catch (e) { /* Google أحياناً بترفضها («Invalid data updating form»): بتطلع رسالة Google العادية إنه الفورم مسكّر */ }
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
  try {
    var parents = DriveApp.getFileById(ss.getId()).getParents();
    if (parents.hasNext()) DriveApp.getFileById(form.getId()).moveTo(parents.next());
  } catch (e) { /* بيضل بـ «ملفاتي» */ }
  var url = form.getPublishedUrl(), short = url;
  try { short = form.shortenFormUrl(url); } catch (e) { /* الرابط الطويل بيشتغل */ }
  return {name: name, short: short, url: url, edit: form.getEditUrl(), n: paper.length, total: total};
}
function linksSheet(ss) {
  var sh = ss.getSheetByName(LINKS_SHEET) || ss.insertSheet(LINKS_SHEET);
  sh.clear();
  sh.setRightToLeft(true);
  return sh;
}
function writeLinks(ss, made, notes) {
  var rows = [["النموذج", "رابط الطلاب (اكتبه على السبورة)", "رابط الطلاب الكامل", "رابط التعديل (للأستاذ بس)", "عدد الأسئلة", "مجموع العلامات"]];
  made.forEach(function (m) { rows.push([m.name, m.short, m.url, m.edit, m.n, m.total]); });
  rows.push(["", "", "", "", "", ""]);
  MANUAL_STEPS.split("\n").concat(notes.length ? ["", "ملاحظات:"].concat(notes) : []).forEach(function (l) { rows.push([l, "", "", "", "", ""]); });
  linksSheet(ss).getRange(1, 1, rows.length, 6).setValues(rows);
}
function fail(ss, errs) {
  var msg = "ما انعمل الفورم. صلّح هالأشياء بملف الأسئلة (أو بالإعدادات فوق بالبرنامج) وشغّله مرة ثانية:\n• " + errs.join("\n• ");
  var rows = [["ما انعمل الفورم: صلّح هالأشياء وشغّل البرنامج مرة ثانية"]].concat(errs.map(function (e) { return [e]; }));
  linksSheet(ss).getRange(1, 1, rows.length, 1).setValues(rows);
  if (!say(msg)) throw new Error(msg);         // من محرّر البرنامج: الخطأ بيطلع بالأحمر تحت
}
function say(msg) {
  Logger.log(msg);
  try { SpreadsheetApp.getUi().alert(msg); return true; } catch (e) { return false; }
}
